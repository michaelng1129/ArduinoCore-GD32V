# Arduino platform definition for GD32VW55x
name=GD32VW55x Arduino Core
version=0.1.0

# Toolchain
# For manual install (no Boards Manager package yet), use the SDK-bundled
# toolchain shipped alongside this repo (not in git - too large, keep it
# next to your checkout):
#   {runtime.platform.path}/system/gd32vw55x/sdk/tools/gd32vw55x_toolchain_windows/bin/
# NOTE: that folder name is Windows-only. A cross-platform tools/ layout
# can replace this when the Boards Manager package is set up.
compiler.path={runtime.platform.path}/system/gd32vw55x/sdk/tools/gd32vw55x_toolchain_windows/bin/
compiler.c.cmd=riscv-nuclei-elf-gcc
compiler.c.flags=-c -g -Os -march=rv32imac -mabi=ilp32 -mcmodel=medlow -mno-save-restore -msmall-data-limit=8 -ffunction-sections -fdata-sections -Wall -Wextra -Wno-unused-parameter -MMD
compiler.cpp.cmd=riscv-nuclei-elf-g++
compiler.cpp.flags=-c -g -Os -march=rv32imac -mabi=ilp32 -mcmodel=medlow -mno-save-restore -msmall-data-limit=8 -ffunction-sections -fdata-sections -fpermissive -fno-exceptions -fno-rtti -std=gnu++17 -MMD
compiler.S.cmd=riscv-nuclei-elf-gcc
compiler.S.flags=-c -g -x assembler-with-cpp -march=rv32imac -mabi=ilp32 -mcmodel=medlow
compiler.ar.cmd=riscv-nuclei-elf-ar
compiler.ar.flags=rcs
compiler.elf2bin.cmd=riscv-nuclei-elf-objcopy
compiler.elf2bin.flags=-O binary
compiler.size.cmd=riscv-nuclei-elf-size

# SDK paths
build.system.path={runtime.platform.path}/system/gd32vw55x/sdk
build.sdk.msdk.path={build.system.path}/MSDK
build.sdk.config.path={build.system.path}/config
build.sdk.rom.path={build.system.path}/ROM-EXPORT
build.sdk.periph.include={build.sdk.msdk.path}/plf/GD32VW55x_standard_peripheral/Include
build.sdk.riscv.include={build.sdk.msdk.path}/plf/riscv
build.sdk.nmsis.include={build.sdk.riscv.include}/NMSIS/Core/Include

# Arduino core, board variant, GD32 standard peripheral, NMSIS and SDK headers
compiler.c.includes=-I{runtime.platform.path}/cores/{build.core} -I{runtime.platform.path}/variants/{build.variant} -I{build.sdk.config.path} -I{build.sdk.msdk.path}/app -I{build.sdk.msdk.path}/plf/src -I{build.sdk.msdk.path}/plf/src/uart -I{build.sdk.msdk.path}/plf/src/time -I{build.sdk.msdk.path}/plf/riscv/gd32vw55x -I{build.sdk.periph.include} -I{build.sdk.riscv.include} -I{build.sdk.nmsis.include} -I{build.sdk.msdk.path}/rtos/FreeRTOS/Source/include -I{build.sdk.msdk.path}/rtos/FreeRTOS/config -I{build.sdk.msdk.path}/rtos/FreeRTOS/Source/portable/GCC/RISC-V -I{build.sdk.msdk.path}/rtos/rtos_wrapper -I{build.sdk.rom.path}/bootloader -I{build.sdk.rom.path}/halcomm -I{build.sdk.rom.path}/mbedtls-2.17.0-rom/include -I{build.sdk.msdk.path}/wifi_manager -I{build.sdk.msdk.path}/ble/app -I{build.sdk.msdk.path}/util/include -I{build.sdk.msdk.path}/macsw/export
compiler.c.extra_flags={compiler.c.includes} {build.extra_flags}
compiler.cpp.extra_flags={compiler.c.includes} {build.extra_flags} -DARDUINO=10810 -DARDUINO_{build.board} -DARDUINO_ARCH_GD32VW55X
compiler.S.extra_flags={compiler.c.includes} {build.extra_flags}

# Link against the SDK libraries shipped with this repository.
compiler.ldscript={runtime.platform.path}/variants/{build.variant}/{build.ldscript}
compiler.ldflags=-T{compiler.ldscript} -march=rv32imac -mabi=ilp32 -mcmodel=medlow -nostartfiles -Wl,--gc-sections -Wl,--no-warn-rwx-segments -Wl,-Map,{build.path}/{build.project_name}.map
compiler.libraries.ldflags=-L{build.sdk.msdk.path}/lib -Wl,--start-group -lwifi -lrf -lble -lwpas -Wl,--end-group -lm -lgcc

# Arduino build recipes
recipe.c.o.pattern="{compiler.path}{compiler.c.cmd}" {compiler.c.flags} {compiler.c.extra_flags} -o "{build.path}/{object_file}" "{source_file}"
recipe.cpp.o.pattern="{compiler.path}{compiler.cpp.cmd}" {compiler.cpp.flags} {compiler.cpp.extra_flags} -o "{build.path}/{object_file}" "{source_file}"
recipe.S.o.pattern="{compiler.path}{compiler.S.cmd}" {compiler.S.flags} {compiler.S.extra_flags} -o "{build.path}/{object_file}" "{source_file}"
recipe.ar.pattern="{compiler.path}{compiler.ar.cmd}" {compiler.ar.flags} "{build.path}/{archive_file}" {object_files}
recipe.c.combine.pattern="{compiler.path}{compiler.c.cmd}" -o "{build.path}/{build.project_name}.elf" {object_files} "{build.path}/{archive_file}" {compiler.ldflags} {compiler.libraries.ldflags}
recipe.objcopy.bin.pattern="{compiler.path}{compiler.elf2bin.cmd}" {compiler.elf2bin.flags} "{build.path}/{build.project_name}.elf" "{build.path}/{build.project_name}.bin"
recipe.size.pattern="{compiler.path}{compiler.size.cmd}" -B "{build.path}/{build.project_name}.elf"
recipe.size.regex=^(?:\.text|\.data|\.rodata|\.bss)\s+[0-9]+
recipe.size.regex.data=^\.data\s+([0-9]+)
recipe.size.regex.bss=^\.bss\s+([0-9]+)

# OpenOCD upload (requires the GigaDevice OpenOCD fork - stock OpenOCD has no
# gd32vw55x flash driver). The vendor GD-Link config is shipped in this repo.
# The Arduino app is linked at 0x0800A000 (OTA slot, see variants/*/ldscript.ld
# and AN154 Table 6-1), so the binary is programmed there, NOT at 0x08000000.
# The MBL bootloader at 0x08000000 must already be on the board (it is on
# stock START boards); it boots the app from the OTA slot.
tools.openocd.path={runtime.platform.path}/tools/openocd
tools.openocd.cmd.path={path}/bin/openocd
tools.openocd.cfg.path={runtime.platform.path}/system/gd32vw55x/sdk/MSDK/projects/eclipse/msdk/openocd_gdlink.cfg
tools.openocd.upload.params.verbose=-d1
tools.openocd.upload.pattern="{cmd.path}" {upload.params.verbose} -f "{tools.openocd.cfg.path}" -c "program {build.path}/{build.project_name}.bin verify reset exit 0x0800A000"
